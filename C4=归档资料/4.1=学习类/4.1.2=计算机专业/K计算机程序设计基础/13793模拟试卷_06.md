using namespace std;
void reversePart(int a[],int n) {

        int t=a[i]; a[i]=a[n-1-i]; a[n-1-i]=t;
    }
}
int main()
{
    int a[4]={1,2,3,4};

    cout << a[0] << ' ' << a[3] << '\n';
